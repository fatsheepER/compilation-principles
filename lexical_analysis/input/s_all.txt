int helper(int x, int y) {
    return x + y;
}

void main() {
    int a = 10, b = 077, c = 0x1F;
    int arr[3];
    int mask = 3;
    char ch = 'a';
    char nl = '\n';
    char quote = '\'';
    char *msg = "hello\tworld";
    int *ptr;

    // single line comment
    /* block comment */

    a++;
    b--;
    a += 1;
    b -= 2;
    c *= 3;
    c /= 4;
    c %= 5;

    if (a < b && b != 0) {
        a = helper(a, b);
    } else {
        a = a > c ? a : c;
    }

    while (a <= c || !mask) {
        a = a << 1;
        c = c >> 1;
        mask = ~mask;
        mask = (a & b) | (b ^ c);
        break;
    }

    do {
        arr[0] = a;
        arr[1] = b;
        arr[2] = c;
        continue;
    } while (arr[0] >= arr[1]);

    ptr->value = arr[1];
    ptr->next = ptr;
    a = ptr->value + arr[2];
    a = a - b * c / 2 % 3;
    a = a + b;
    a = a . b;

    return;
}
